cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
project(DMAgent VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "DMAgent requires Windows and the MSVC C++ toolchain.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This source package builds the x64 Agent.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
add_library(dm_common STATIC src/common/win.cpp src/common/model.cpp src/common/shared_memory.cpp)
target_include_directories(dm_common PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
target_link_libraries(dm_common PUBLIC advapi32 ole32 bcrypt)
add_executable(DmTelemetryAgent WIN32
  src/agent/main.cpp src/agent/controller.cpp src/agent/protocol.cpp
  src/agent/serial.cpp src/agent/storage.cpp src/agent/energy.cpp resources/agent.rc)
target_include_directories(DmTelemetryAgent PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/resources")
target_link_libraries(DmTelemetryAgent PRIVATE dm_common comctl32 shell32 setupapi user32 gdi32)
target_link_options(DmTelemetryAgent PRIVATE /MANIFEST:NO)
install(TARGETS DmTelemetryAgent RUNTIME DESTINATION .)
install(FILES docs/PORTABLE_README.txt DESTINATION . RENAME README.txt)
install(FILES LICENSE DESTINATION .)
install(FILES docs/USAGE.md docs/IPC.md docs/ENERGY.md docs/PROTOCOL.md docs/THIRD_PARTY.md DESTINATION docs)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-LICENSE.MIT)
install(FILES viewer/dist/DM-Record-Viewer.html viewer/dist/build-manifest.json viewer/USAGE.md DESTINATION viewer)
# Build/install only. No application, browser, device or test is launched.
